#!/usr/bin/env python3
"""Create named RGB signal presets from an RGB + CIE xy table."""

from __future__ import annotations

import argparse
import colorsys
import csv
import json
import math
import re
from pathlib import Path


ROW_RE = re.compile(
    r"^\s*([0-9A-Fa-f]{1,2})\s+([0-9A-Fa-f]{1,2})\s+"
    r"([0-9A-Fa-f]{1,2})\s+([0-9]*\.?[0-9]+)\s+([0-9]*\.?[0-9]+)\s*$"
)


def xy_hue(x, y):
    """Convert CIE xy to an approximate display hue; xy is the name source."""
    big_x = x / y
    big_y = 1.0
    big_z = max(0.0, (1.0 - x - y) / y)
    red = max(0.0, 3.2404542 * big_x - 1.5371385 * big_y - 0.4985314 * big_z)
    green = max(0.0, -0.9692660 * big_x + 1.8760108 * big_y + 0.0415560 * big_z)
    blue = max(0.0, 0.0556434 * big_x - 0.2040259 * big_y + 1.0572252 * big_z)
    maximum = max(red, green, blue, 1e-9)
    return colorsys.rgb_to_hsv(
        red / maximum, green / maximum, blue / maximum
    )[0] * 360.0


def hue_name(hue):
    ranges = (
        (15.0, "红色"), (30.0, "朱红色"), (45.0, "橙色"),
        (60.0, "琥珀色"), (80.0, "黄色"), (110.0, "黄绿色"),
        (150.0, "绿色"), (180.0, "青绿色"), (205.0, "青色"),
        (225.0, "天蓝色"), (255.0, "蓝色"), (275.0, "蓝紫色"),
        (300.0, "紫色"), (330.0, "品红色"), (345.0, "玫红色"),
        (360.0, "红色"),
    )
    for upper, name in ranges:
        if hue < upper:
            return name
    return "红色"


def parse_rows(path):
    rows = []
    for line_number, line in enumerate(
        Path(path).read_text(encoding="utf-8-sig").splitlines(), 1
    ):
        if not line.strip() or line.lstrip().startswith("R"):
            continue
        match = ROW_RE.match(line)
        if match is None:
            raise ValueError("无法识别第 {0} 行颜色数据".format(line_number))
        red, green, blue = (int(match.group(i), 16) for i in range(1, 4))
        x, y = float(match.group(4)), float(match.group(5))
        if x < 0.0 or y <= 0.0 or x + y > 1.0:
            raise ValueError("第 {0} 行的 CIE xy 无效".format(line_number))
        rows.append((red, green, blue, x, y))
    if not rows:
        raise ValueError("没有找到 RGB/xy 数据")
    return rows


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", required=True)
    parser.add_argument("--output", required=True)
    parser.add_argument("--audit-csv", required=True)
    args = parser.parse_args()

    rows = parse_rows(args.input)
    items = []
    audit = []
    for index, (red, green, blue, x, y) in enumerate(rows, 1):
        base_name = hue_name(xy_hue(x, y))
        visible_name = "{0} {1:03d}".format(base_name, index)
        items.append(
            {
                "name": "{0}|x={1:.4f} y={2:.4f}".format(visible_name, x, y),
                "values": {"RGB_R": red, "RGB_G": green, "RGB_B": blue},
            }
        )
        audit.append(
            {
                "index": index,
                "name": visible_name,
                "R_hex": "{0:02X}".format(red),
                "G_hex": "{0:02X}".format(green),
                "B_hex": "{0:02X}".format(blue),
                "R": red, "G": green, "B": blue,
                "x": "{0:.4f}".format(x), "y": "{0:.4f}".format(y),
            }
        )

    overlay = {
        "signal_presets": {
            "group_name": "xy 色度色板 · {0} 色".format(len(items)),
            "items": items,
        }
    }
    Path(args.output).write_text(
        json.dumps(overlay, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )
    with Path(args.audit_csv).open("w", encoding="utf-8-sig", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(audit[0]))
        writer.writeheader()
        writer.writerows(audit)
    print(
        "已生成 {0} 个 xy 命名颜色，共 {1} 页".format(
            len(rows), int(math.ceil(len(rows) / 30.0))
        )
    )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
